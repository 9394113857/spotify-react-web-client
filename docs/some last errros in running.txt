✅ REPLACE WITH THIS (FIXED VERSION)
const base64encode = (input: ArrayBuffer) => {
  const bytes = new Uint8Array(input);

  let binary = '';
  for (let i = 0; i < bytes.length; i++) {
    binary += String.fromCharCode(bytes[i]);
  }

  return btoa(binary)
    .replace(/=/g, '')
    .replace(/\+/g, '-')
    .replace(/\//g, '_');
};
